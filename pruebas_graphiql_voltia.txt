# =====================================================================
#  1. CONSULTAS (QUERIES)
# =====================================================================

# 1.1 Todas las lavadoras con sus mantenimientos y campos calculados.
query TodasLasLavadoras {
  lavadoras {
    codigo
    marca
    precioBase
    fechaFabricacion
    capacidadCargaKilos
    velocidadCentrifugadoRPM
    tecnologiaInverter
    minutosLavado
    precioFinal
    cantidadMantenimientos
    mantenimientosPendientes
    costoMantenimientos
    mantenimientos {
      id
      descripcion
      tecnico
      costo
      fecha
      completado
    }
  }
}

# 1.2 Filtrar por marca y por tecnologia inverter (filtro del servidor).
query FiltrarPorMarcaEInverter {
  lavadorasPorFiltro(marca: "lg", tecnologiaInverter: true) {
    codigo
    marca
    tecnologiaInverter
    precioFinal
  }
}

# 1.3 Filtrar por tenencia de mantenimientos:
#     conMantenimientos: true  -> solo las que tienen uno o mas
#     conMantenimientos: false -> solo las que no tienen ninguno
query FiltrarPorMantenimientos {
  conMantenimientos: lavadorasPorFiltro(conMantenimientos: true) {
    codigo
    marca
    cantidadMantenimientos
  }
  sinMantenimientos: lavadorasPorFiltro(conMantenimientos: false) {
    codigo
    marca
    cantidadMantenimientos
  }
}

# 1.4 Una lavadora por codigo.
query LavadoraPorCodigo {
  lavadoraPorCodigo(codigo: 101) {
    codigo
    marca
    precioFinal
    mantenimientos {
      id
      descripcion
    }
  }
}

# 1.5 Una lavadora por uno o dos criterios (codigo y/o marca).
query LavadoraPorCriterios {
  lavadoraPorCriterios(marca: "Samsung") {
    codigo
    marca
    minutosLavado
    mantenimientos {
      id
      descripcion
      completado
    }
  }
}

# 1.6 Mantenimientos de una lavadora (el detalle).
query MantenimientosDeUnaLavadora {
  mantenimientos(codigoLavadora: 101) {
    id
    descripcion
    tecnico
    costo
    fecha
    completado
  }
}

# 1.7 Filtrar mantenimientos por tecnico y por estado (filtro del servidor).
query MantenimientosFiltrados {
  mantenimientosPorFiltro(codigoLavadora: 101, tecnico: "carlos", completado: false) {
    id
    descripcion
    tecnico
    completado
  }
}

# 1.8 Un mantenimiento por su id dentro de la lavadora.
query MantenimientoPorId {
  mantenimientoPorId(codigoLavadora: 104, id: 2) {
    id
    descripcion
    costo
    completado
  }
}

# 1.9 Version de los datos: sube cada vez que un cliente modifica algo.
query VersionDeLosDatos {
  versionDatos
}


# =====================================================================
#  2. MUTACIONES
# =====================================================================

# 2.1 Crear una lavadora (nace con cero mantenimientos).
mutation CrearLavadora($entrada: LavadoraInput!) {
  addLavadora(input: $entrada) {
    codigo
    marca
    precioFinal
    cantidadMantenimientos
    mantenimientos {
      id
    }
  }
}
# VARIABLES:
# {
#   "entrada": {
#     "codigo": 105,
#     "marca": "Mabe",
#     "precioBase": 1120000,
#     "fechaFabricacion": "2026-01-20T09:30:00",
#     "capacidadCargaKilos": 17.5,
#     "velocidadCentrifugadoRPM": 1300,
#     "tecnologiaInverter": true,
#     "minutosLavado": 50
#   }
# }

# 2.2 Actualizar una lavadora. Solo se manda lo que cambia.
#     El servidor primero BUSCA la lavadora; si no existe, responde error.
mutation ActualizarLavadora($codigo: Int!, $entrada: LavadoraUpdateInput!) {
  actualizarLavadora(codigo: $codigo, input: $entrada) {
    codigo
    marca
    precioBase
    precioFinal
  }
}
# VARIABLES:
# {
#   "codigo": 105,
#   "entrada": { "precioBase": 1099000, "tecnologiaInverter": false }
# }

# 2.3 Eliminar una lavadora (busqueda previa y borrado del detalle incluido).
mutation EliminarLavadora($codigo: Int!) {
  eliminarLavadora(codigo: $codigo)
}
# VARIABLES:
# { "codigo": 105 }

# 2.4 Agregar un mantenimiento a una lavadora existente.
#     El id NO se manda: el servidor calcula el siguiente numero libre.
mutation AgregarMantenimiento($codigoLavadora: Int!, $entrada: MantenimientoInput!) {
  addMantenimiento(codigoLavadora: $codigoLavadora, input: $entrada) {
    id
    descripcion
    tecnico
    costo
    fecha
    completado
  }
}
# VARIABLES:
# {
#   "codigoLavadora": 103,
#   "entrada": {
#     "descripcion": "Revision general de la bomba",
#     "tecnico": "Diana Torres",
#     "costo": 65000,
#     "fecha": "2026-01-15T10:00:00",
#     "completado": false
#   }
# }

# 2.5 Actualizar un mantenimiento (parcial).
mutation ActualizarMantenimiento($codigoLavadora: Int!, $id: Int!, $entrada: MantenimientoUpdateInput!) {
  actualizarMantenimiento(codigoLavadora: $codigoLavadora, id: $id, input: $entrada) {
    id
    descripcion
    completado
  }
}
# VARIABLES:
# {
#   "codigoLavadora": 103,
#   "id": 1,
#   "entrada": { "completado": true, "costo": 70000 }
# }

# 2.6 Eliminar un mantenimiento de una lavadora.
mutation EliminarMantenimiento($codigoLavadora: Int!, $id: Int!) {
  eliminarMantenimiento(codigoLavadora: $codigoLavadora, id: $id)
}
# VARIABLES:
# { "codigoLavadora": 103, "id": 1 }


# =====================================================================
#  3. PRUEBAS DE ERROR (deben responder con un mensaje claro)
# =====================================================================

# 3.1 Codigo de lavadora repetido.
mutation ErrorCodigoRepetido {
  addLavadora(input: {
    codigo: 101
    marca: "Prueba"
    precioBase: 1000000
    fechaFabricacion: "2025-01-01T00:00:00"
    capacidadCargaKilos: 10
    velocidadCentrifugadoRPM: 1000
    tecnologiaInverter: false
    minutosLavado: 30
  }) {
    codigo
  }
}

# 3.2 Fecha con formato invalido.
mutation ErrorFechaInvalida {
  addLavadora(input: {
    codigo: 999
    marca: "Prueba"
    precioBase: 1000000
    fechaFabricacion: "01/01/2025"
    capacidadCargaKilos: 10
    velocidadCentrifugadoRPM: 1000
    tecnologiaInverter: false
    minutosLavado: 30
  }) {
    codigo
  }
}

# 3.3 Lavadora inexistente (la busqueda previa falla).
mutation ErrorLavadoraInexistente {
  actualizarLavadora(codigo: 777, input: { marca: "Nadie" }) {
    codigo
  }
}

# 3.4 Mantenimiento inexistente en una lavadora existente.
mutation ErrorMantenimientoInexistente {
  eliminarMantenimiento(codigoLavadora: 101, id: 999)
}

# 3.5 Consulta sin criterios (debe pedir al menos uno).
query ErrorSinCriterios {
  lavadoraPorCriterios {
    codigo
    marca
  }
}
